import assert from 'node:assert/strict';

// Frozen before reading the production implementation. This result oracle scans
// linearly; it does not execute binary search or infer results from its frames.
export function admit(values, target) {
  assert.ok(Array.isArray(values), 'values must be an Array');
  assert.ok(values.length <= 32, 'at most 32 values');
  assert.ok(Number.isInteger(target) && target >= -9999 && target <= 9999, 'bounded integer target');
  for (let i = 0; i < values.length; i += 1) {
    assert.ok(Object.hasOwn(values, i), 'sparse values are not admitted');
    assert.ok(Number.isInteger(values[i]) && values[i] >= -9999 && values[i] <= 9999, 'bounded integer element');
    if (i > 0) assert.ok(values[i - 1] <= values[i], 'nondecreasing order');
  }
}

export function scanOracle(values, target) {
  admit(values, target);
  let boundary = values.length;
  for (let i = 0; i < values.length; i += 1) {
    if (values[i] >= target) { boundary = i; break; }
  }
  return { boundary, found: boundary < values.length && values[boundary] === target };
}

// A normalized frame is {lo, hi, mid?, relation?}. Every nonterminal frame
// describes one ordering comparison; the final frame has lo===hi and no mid.
// Any production adapter must only map recorded fields, not compute new search
// states, the answer, or the comparison count.
export function verifyTrace(values, target, trace) {
  const expected = scanOracle(values, target);
  assert.ok(trace && Array.isArray(trace.frames) && trace.frames.length >= 1, 'frames including terminal state');
  const { frames } = trace;
  assert.equal(frames[0].lo, 0, 'initial lower bound');
  assert.equal(frames[0].hi, values.length, 'initial upper bound');
  let observedComparisons = 0;
  for (let j = 0; j < frames.length; j += 1) {
    const frame = frames[j];
    assert.ok(Number.isInteger(frame.lo) && Number.isInteger(frame.hi), 'integer boundaries');
    assert.ok(0 <= frame.lo && frame.lo <= frame.hi && frame.hi <= values.length, 'valid half-open interval');
    assert.ok(frame.lo <= expected.boundary && expected.boundary <= frame.hi, 'possible boundary interval is closed');
    for (let i = 0; i < frame.lo; i += 1) assert.ok(values[i] < target, 'discarded prefix is strictly less');
    for (let i = frame.hi; i < values.length; i += 1) assert.ok(values[i] >= target, 'discarded suffix is not less');
    if (frame.lo === frame.hi) {
      assert.equal(j, frames.length - 1, 'only the last frame is terminal');
      assert.equal(frame.mid, undefined, 'terminal membership check is not an ordering comparison');
      assert.equal(frame.relation, undefined, 'terminal state has no ordering relation');
      continue;
    }
    assert.ok(j + 1 < frames.length, 'every comparison has its next state');
    assert.equal(frame.mid, frame.lo + Math.floor((frame.hi - frame.lo) / 2), 'specified midpoint');
    const less = values[frame.mid] < target;
    assert.equal(frame.relation, less ? 'less' : 'not_less', 'ordering relation');
    const next = frames[j + 1];
    assert.equal(next.lo, less ? frame.mid + 1 : frame.lo, 'next lower boundary');
    assert.equal(next.hi, less ? frame.hi : frame.mid, 'next upper boundary');
    assert.ok(next.hi - next.lo < frame.hi - frame.lo, 'strict shrink per ordering comparison');
    observedComparisons += 1;
  }
  const terminal = frames.at(-1);
  assert.equal(terminal.lo, terminal.hi, 'terminated interval');
  assert.equal(trace.boundary, expected.boundary, 'first not-less position from independent linear scan');
  assert.equal(terminal.lo, trace.boundary, 'terminal boundary');
  assert.equal(trace.found, expected.found, 'membership is a separate equality fact');
  assert.equal(trace.orderingComparisons, observedComparisons, 'count only ordering comparisons');
  assert.ok(observedComparisons <= Math.ceil(Math.log2(values.length + 1)), 'binary-search comparison bound');
  return expected;
}

export function* smallSortedArrays() {
  const alphabet = [-2, -1, 0, 1, 2];
  function* grow(prefix, start) {
    yield [...prefix];
    if (prefix.length === 7) return;
    for (let i = start; i < alphabet.length; i += 1) yield* grow([...prefix, alphabet[i]], i);
  }
  yield* grow([], 0);
}

export const smallTargets = Object.freeze([-3, -2, -1, 0, 1, 2, 3]);
const frame = (lo, hi, mid, relation) => mid === undefined ? { lo, hi } : { lo, hi, mid, relation };
const manual = (name, values, target, boundary, found, orderingComparisons, frames) => ({ name, values, target, trace: { boundary, found, orderingComparisons, frames } });

// These traces are hand-worked constants, not generated by a search loop.
export const manualCases = [
  manual('empty', [], 0, 0, false, 0, [frame(0, 0)]),
  manual('singleton match', [2], 2, 0, true, 1, [frame(0, 1, 0, 'not_less'), frame(0, 0)]),
  manual('first duplicate', [1, 2, 2, 2, 4], 2, 1, true, 3, [frame(0, 5, 2, 'not_less'), frame(0, 2, 1, 'not_less'), frame(0, 1, 0, 'less'), frame(1, 1)]),
  manual('absent between runs', [1, 3, 3, 7], 4, 3, false, 2, [frame(0, 4, 2, 'less'), frame(3, 4, 3, 'not_less'), frame(3, 3)]),
  manual('below minimum', [1, 3, 3, 7], 0, 0, false, 3, [frame(0, 4, 2, 'not_less'), frame(0, 2, 1, 'not_less'), frame(0, 1, 0, 'not_less'), frame(0, 0)]),
  manual('above maximum', [1, 3, 3, 7], 8, 4, false, 2, [frame(0, 4, 2, 'less'), frame(3, 4, 3, 'less'), frame(4, 4)]),
  manual('all duplicates', [-5, -5, -5, -5], -5, 0, true, 3, [frame(0, 4, 2, 'not_less'), frame(0, 2, 1, 'not_less'), frame(0, 1, 0, 'not_less'), frame(0, 0)]),
  manual('upper integer bound', [-9999, 0, 9999], 9999, 2, true, 2, [frame(0, 3, 1, 'less'), frame(2, 3, 2, 'not_less'), frame(2, 2)]),
  manual('lower integer bound', [-9999, 0, 9999], -9999, 0, true, 2, [frame(0, 3, 1, 'not_less'), frame(0, 1, 0, 'not_less'), frame(0, 0)]),
  manual('32 elements, absent low edge', Array.from({ length: 32 }, (_, i) => i), -1, 0, false, 6, [frame(0, 32, 16, 'not_less'), frame(0, 16, 8, 'not_less'), frame(0, 8, 4, 'not_less'), frame(0, 4, 2, 'not_less'), frame(0, 2, 1, 'not_less'), frame(0, 1, 0, 'not_less'), frame(0, 0)]),
  manual('32 elements, equal low edge', Array.from({ length: 32 }, (_, i) => i), 0, 0, true, 6, [frame(0, 32, 16, 'not_less'), frame(0, 16, 8, 'not_less'), frame(0, 8, 4, 'not_less'), frame(0, 4, 2, 'not_less'), frame(0, 2, 1, 'not_less'), frame(0, 1, 0, 'not_less'), frame(0, 0)]),
];

export function invalidCases() {
  return [
    { name: 'unsorted', values: [1, 0], target: 0 },
    { name: 'fractional element', values: [0.5], target: 0 },
    { name: 'NaN element', values: [NaN], target: 0 },
    { name: 'infinite element', values: [Infinity], target: 0 },
    { name: 'low element', values: [-10000], target: 0 },
    { name: 'high element', values: [10000], target: 0 },
    { name: 'too many', values: Array(33).fill(0), target: 0 },
    { name: 'sparse', values: Array(1), target: 0 },
    { name: 'string element', values: ['1'], target: 1 },
    { name: 'not an array', values: { 0: 1, length: 1 }, target: 1 },
    { name: 'fractional target', values: [], target: 0.5 },
    { name: 'NaN target', values: [], target: NaN },
    { name: 'low target', values: [], target: -10000 },
    { name: 'high target', values: [], target: 10000 },
    { name: 'string target', values: [1], target: '1' },
    { name: 'null target', values: [], target: null },
  ];
}

export function selfCheckOracle() {
  const arrays = [...smallSortedArrays()];
  assert.equal(arrays.length, 792);
  for (const item of manualCases) verifyTrace(item.values, item.target, item.trace);
  for (const item of invalidCases()) assert.throws(() => scanOracle(item.values, item.target), undefined, item.name);
  const duplicate = structuredClone(manualCases.find((c) => c.name === 'first duplicate'));
  duplicate.trace.boundary = 2;
  assert.throws(() => verifyTrace(duplicate.values, duplicate.target, duplicate.trace), /first not-less position/);
  const extraEquality = structuredClone(manualCases[1]);
  extraEquality.trace.orderingComparisons += 1;
  assert.throws(() => verifyTrace(extraEquality.values, extraEquality.target, extraEquality.trace), /count only ordering comparisons/);
  const wrongMid = structuredClone(manualCases[2]);
  wrongMid.trace.frames[0].mid = 3;
  assert.throws(() => verifyTrace(wrongMid.values, wrongMid.target, wrongMid.trace), /specified midpoint/);
  const nonshrinking = structuredClone(manualCases[1]);
  nonshrinking.trace.frames.splice(1, 0, structuredClone(nonshrinking.trace.frames[0]));
  assert.throws(() => verifyTrace(nonshrinking.values, nonshrinking.target, nonshrinking.trace), /next upper boundary|strict shrink/);
  const wrongSuffix = structuredClone(manualCases[2]);
  wrongSuffix.trace.frames[0].hi = 2;
  assert.throws(() => verifyTrace(wrongSuffix.values, wrongSuffix.target, wrongSuffix.trace), /initial upper bound/);
  assert.deepEqual(scanOracle([-0, 0, 1], -0), { boundary: 0, found: true });
  return { smallSortedArrays: arrays.length, exhaustiveCandidateCases: arrays.length * smallTargets.length,
    handWorkedTraces: manualCases.length, invalidControls: invalidCases().length, deliberatelyBadTraceControls: 5,
    productionImplementationRead: false, productionAdapterPresent: false };
}
