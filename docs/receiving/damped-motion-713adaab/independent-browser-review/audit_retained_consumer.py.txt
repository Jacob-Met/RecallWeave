# Audit retained values against an elementary exact critical-case formula only.
# This does not import the product, invoke its math core, or rerun the browser.

    t=state["t"];assert 0<=t<=4
    x=(.25-.75*t)*math.exp(-t);v=(-1+.75*t)*math.exp(-t)
    expected={"x":x,"v":v,"a":-x-2*v,"springForce":-x,"dampingForce":-2*v,
              "kineticEnergy":.5*v*v,"springEnergy":.5*x*x,
              "mechanicalEnergy":.5*(v*v+x*x),"energyRate":-2*v*v,
              "energyTransferred":.53125-.5*(v*v+x*x)}
    for name,value in expected.items():
        assert math.isfinite(state[name])
        delta=abs(state[name]-value)
        maximum_error=max(maximum_error,delta)
        assert delta<=2e-11+2e-11*abs(value),(t,name,delta)
