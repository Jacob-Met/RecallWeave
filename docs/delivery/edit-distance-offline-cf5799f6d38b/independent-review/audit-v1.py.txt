class Doc(HTMLParser):
    def __init__(self):
        super().__init__();self.refs=[];self.ids=set();self.scripts=[];self.styles=[];self.active=None;self.attrs=None;self.data=[]
    def handle_starttag(self,tag,attrs):
        a=dict(attrs)
        if 'id' in a:self.ids.add(a['id'])
        for k in ('href','src','poster','action','data','srcset'):
            if k in a:self.refs.append(dict(tag=tag,attribute=k,target=a[k]))
        if tag in ('script','style'):self.active=tag;self.attrs=a;self.data=[]
    def handle_data(self,s):
        if self.active:self.data.append(s)
    def handle_endtag(self,tag):
        if tag==self.active:
            (self.scripts if tag=='script' else self.styles).append((self.attrs,''.join(self.data)))
            self.active=None
