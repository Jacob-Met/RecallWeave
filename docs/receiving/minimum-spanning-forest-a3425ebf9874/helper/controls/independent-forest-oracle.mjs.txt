import assert from 'node:assert/strict';

export function componentsOf(n, edges, ids = edges.map((_, i) => i)) {
  const adjacency = Array.from({ length: n }, () => []);
  for (const id of ids) {
    const edge = edges[id];
    adjacency[edge.from].push(edge.to);
    adjacency[edge.to].push(edge.from);
  }
  const seen = new Set(), groups = [];
  for (let start = 0; start < n; start++) {
    if (seen.has(start)) continue;
    const queue = [start], group = [];
    seen.add(start);
    for (let i = 0; i < queue.length; i++) {
      const vertex = queue[i];
      group.push(vertex);
      for (const next of adjacency[vertex]) {
        if (!seen.has(next)) { seen.add(next); queue.push(next); }
      }
    }
    group.sort((a, b) => a - b);
    groups.push(group);
  }
  return groups;
}

// Enumerate candidate edge subsets of exactly the required forest rank.
// Connectivity is recomputed by graph traversal; no greedy result or disjoint-set
// implementation supplies the optimum. Subsets must span every input component.
export function minimumForestWeight(n, edges) {
  const components = componentsOf(n, edges);
  const rank = n - components.length;
  if (rank === 0) return { weight: 0, ids: [], enumerated: 1, components };
  const lowerBound = edges.map(e => e.weight).sort((a, b) => a - b)
    .slice(0, rank).reduce((sum, weight) => sum + weight, 0);
  let best = Infinity, bestIds = [], enumerated = 0;
  const chosen = [];
  function visit(start, remaining) {
    if (best === lowerBound) return;
    if (remaining === 0) {
      enumerated++;
      if (componentsOf(n, edges, chosen).length !== components.length) return;
      const weight = chosen.reduce((sum, id) => sum + edges[id].weight, 0);
      if (weight < best) { best = weight; bestIds = [...chosen]; }
      return;
    }
    for (let id = start; id <= edges.length - remaining; id++) {
      chosen.push(id); visit(id + 1, remaining - 1); chosen.pop();
      if (best === lowerBound) return;
    }
  }
  visit(0, rank);
  assert.ok(Number.isFinite(best), 'Every admitted graph has a spanning forest');
  return { weight: best, ids: bestIds, enumerated, components };
}

export function assertDeepFrozen(value, seen = new Set()) {
  if (value === null || typeof value !== 'object' || seen.has(value)) return;
  seen.add(value);
  assert.equal(Object.isFrozen(value), true, 'Every returned object/array must be frozen');
  for (const child of Object.values(value)) assertDeepFrozen(child, seen);
}

export function inspectTrace(n, edges, trace) {
  assert.equal(trace.format, 'recallweave.minimum-spanning-forest-trace/1');
  assert.equal(trace.vertexCount, n);
  assert.deepEqual(trace.edges, edges.map((edge, id) =>
    ({ id, from: edge.from, to: edge.to, weight: edge.weight })));
  assert.notEqual(trace.edges, edges);
  for (let i = 0; i < edges.length; i++) assert.notEqual(trace.edges[i], edges[i]);
  const order = edges.map((_, id) => id).sort((a, b) =>
    edges[a].weight - edges[b].weight || a - b);
  assert.deepEqual(trace.orderedEdgeIds, order);
  assert.equal(trace.steps.length, edges.length, 'Decide every edge, including final cycle rejections');
  const accepted = [];
  let total = 0;
  for (let i = 0; i < order.length; i++) {
    const edgeId = order[i], edge = edges[edgeId], step = trace.steps[i];
    const before = componentsOf(n, edges, accepted);
    const sameComponent = before.some(group => group.includes(edge.from) && group.includes(edge.to));
    assert.equal(step.number, i + 1);
    assert.equal(step.edgeId, edgeId);
    assert.equal(step.from, edge.from);
    assert.equal(step.to, edge.to);
    assert.ok(Object.is(step.weight, edge.weight), 'Retain the input edge weight');
    assert.equal(step.accepted, !sameComponent);
    assert.deepEqual(step.beforeComponents, before);
    if (!sameComponent) { accepted.push(edgeId); total += edge.weight; }
    const after = componentsOf(n, edges, accepted);
    assert.deepEqual(step.afterComponents, after);
    assert.deepEqual(step.acceptedEdgeIds, accepted);
    assert.equal(step.totalWeight, total);
    assert.equal(after.length, before.length - (sameComponent ? 0 : 1));
    assert.equal(accepted.length, n - after.length, 'Accepted edges are acyclic at every step');
  }
  const optimum = minimumForestWeight(n, edges);
  const finalGroups = componentsOf(n, edges, accepted);
  assert.deepEqual(trace.forestEdgeIds, accepted);
  assert.deepEqual(trace.components, finalGroups);
  assert.deepEqual(finalGroups, optimum.components, 'The forest spans all input components, including isolates');
  assert.equal(trace.totalWeight, total);
  assert.equal(total, optimum.weight, 'The final forest has independently enumerated minimum weight');
  assert.equal(trace.connected, finalGroups.length === 1);
  assert.equal(trace.decisions, edges.length);
  assertDeepFrozen(trace);
  return optimum.enumerated;
}

export function* exhaustiveGraphs() {
  for (let n = 1; n <= 4; n++) {
    const pairs = [];
    for (let from = 0; from < n; from++)
      for (let to = from + 1; to < n; to++) pairs.push([from, to]);
    for (let encoding = 0; encoding < 4 ** pairs.length; encoding++) {
      let code = encoding;
      const edges = [];
      for (let i = 0; i < pairs.length; i++) {
        const choice = code % 4; code = Math.floor(code / 4);
        if (!choice) continue;
        let [from, to] = pairs[i];
        if ((encoding + i) % 2) [from, to] = [to, from];
        edges.push({ from, to, weight: [-2, 0, 3][choice - 1] });
      }
      if (encoding % 2) edges.reverse();
      yield { n, edges, label: 'exhaustive-n' + n + '-' + encoding };
    }
  }
}

export function* seededGraphs() {
  let state = 0x65c93f17;
  const next = () => {
    state ^= state << 13; state ^= state >>> 17; state ^= state << 5;
    return state >>> 0;
  };
  for (let index = 0; index < 96; index++) {
    const n = 5 + index % 4, pairs = [];
    for (let from = 0; from < n; from++)
      for (let to = from + 1; to < n; to++) pairs.push([from, to]);
    for (let i = pairs.length - 1; i > 0; i--) {
      const j = next() % (i + 1); [pairs[i], pairs[j]] = [pairs[j], pairs[i]];
    }
    const count = next() % (Math.min(12, pairs.length) + 1);
    const edges = pairs.slice(0, count).map(([a, b]) => {
      const reverse = next() % 2;
      return { from: reverse ? b : a, to: reverse ? a : b,
        weight: [-99, -4, -1, 0, 0, 2, 7, 99][next() % 8] };
    });
    yield { n, edges, label: 'seeded-' + index };
  }
}

function freeze(value) {
  if (value && typeof value === 'object') {
    for (const child of Object.values(value)) freeze(child);
    Object.freeze(value);
  }
  return value;
}

export function selfCheck() {
  let graphs = 0;
  for (const item of handGraphs) {
    assert.equal(minimumForestWeight(item.n, item.edges).weight, item.weight, item.label);
    graphs++;
  }
  const edges = [{ from: 1, to: 0, weight: 7 }];
  const good = {
    format: 'recallweave.minimum-spanning-forest-trace/1', vertexCount: 2,
    edges: [{ id: 0, from: 1, to: 0, weight: 7 }], orderedEdgeIds: [0],
    steps: [{ number: 1, edgeId: 0, from: 1, to: 0, weight: 7, accepted: true,
      beforeComponents: [[0], [1]], afterComponents: [[0, 1]], acceptedEdgeIds: [0], totalWeight: 7 }],
    forestEdgeIds: [0], totalWeight: 7, components: [[0, 1]], connected: true, decisions: 1
  };
  inspectTrace(2, edges, freeze(structuredClone(good)));
  const mutations = [
    x => { x.format = 'wrong'; }, x => { x.decisions = 0; },
    x => { x.steps = []; }, x => { x.steps[0].accepted = false; },
    x => { x.steps[0].beforeComponents = [[0, 1]]; },
    x => { x.steps[0].totalWeight = 6; }, x => { x.totalWeight = 6; },
    x => { x.edges[0].from = 0; }, x => { x.connected = false; }
  ];
  for (const mutate of mutations) {
    const bad = structuredClone(good); mutate(bad); freeze(bad);
    assert.throws(() => inspectTrace(2, edges, bad), undefined, 'Known bad trace must be refused by checker');
  }
  assert.throws(() => inspectTrace(2, edges, structuredClone(good)));
  assert.equal([...exhaustiveGraphs()].length, 4165);
  return { handOptima: graphs, admittedManualTrace: 1, refusedBadTraceControls: mutations.length + 1,
    exhaustiveGraphDefinitions: 4165, seededGraphDefinitions: 96 };
}
if (process.argv.includes('--self-test')) console.log(JSON.stringify(selfCheck()));
